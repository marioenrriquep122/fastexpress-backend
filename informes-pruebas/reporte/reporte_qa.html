2Algoritmo principal de cálculo de tarifa (BR-01 a BR-04).
3

5Es pura: no toca base de datos ni HTTP, por eso se prueba en milisegundos.

14 recargo = subtotal * % envío ("sobre la tarifa subtotal")
15 total = subtotal - descuento + recargo
16"""

19
20from principal.constantes.cotizacion import (

22 COSTO_BASE,
23 DESCUENTO_FRECUENTE,
24 DESCUENTO_NUEVO,
25 DESCUENTO_VIP,

30 TARIFA_KG_LIGERO,
31 TARIFA_KG_PESADO,

33 TipoEnvio,
34)

40)
41
42
43def redondear(valor: Decimal) -> Decimal:
44 """Redondea dinero a 2 decimales, mitad hacia arriba (0.005 -> 0.01)."""
45 return valor.quantize(CENTAVO, rounding=ROUND_HALF_UP)
46

48def calcular_cotizacion(peso, tipo_cliente: str, tipo_envio: str) -> dict:
49 """
50 Aplica BR-01 a BR-04 y devuelve el desglose de la cotización.
51
52 Raises:
53 PesoInvalidoError: peso <= 0
54 LimiteExcedidoError: peso > 20.00 kg
55 TipoClienteInvalidoError / TipoEnvioInvalidoError: tipo desconocido
56 """
57 peso = Decimal(str(peso))
58
59 # ---- BR-02: validación y tramo de peso --------------------------------
60 if peso <= 0: # [N1] D1
61 raise PesoInvalidoError("El peso debe ser mayor a 0 kg.") # [N2]
62
63 if peso > PESO_MAXIMO_KG: # [N3] D2
64 raise LimiteExcedidoError( # [N4]
65 f"Envío rechazado: el peso máximo permitido es {PESO_MAXIMO_KG} kg."
66 )
67
68 if peso <= LIMITE_TRAMO_LIGERO_KG: # [N5] D3
69 tarifa_kg = TARIFA_KG_LIGERO # [N6]
70 else:
71 tarifa_kg = TARIFA_KG_PESADO # [N7]
72
73 # ---- BR-03: descuento por tipo de cliente -----------------------------
74 if tipo_cliente == TipoCliente.NUEVO: # [N8] D4
75 porcentaje_descuento = DESCUENTO_NUEVO # [N9]
76 elif tipo_cliente == TipoCliente.FRECUENTE: # [N10] D5
77 porcentaje_descuento = DESCUENTO_FRECUENTE # [N11]
78 elif tipo_cliente == TipoCliente.VIP: # [N12] D6
79 porcentaje_descuento = DESCUENTO_VIP # [N13]
80 else:
81 raise TipoClienteInvalidoError( # [N14]
82 f"Tipo de cliente inválido: '{tipo_cliente}'."
83 )
84
85 # ---- BR-04: recargo por tipo de envío ---------------------------------
86 if tipo_envio == TipoEnvio.ESTANDAR: # [N15] D7
87 porcentaje_recargo = RECARGO_ESTANDAR # [N16]
88 elif tipo_envio == TipoEnvio.EXPRESS: # [N17] D8
89 porcentaje_recargo = RECARGO_EXPRESS # [N18]
90 else:
91 raise TipoEnvioInvalidoError( # [N19]
92 f"Tipo de envío inválido: '{tipo_envio}'."
93 )
94
95 # ---- BR-01 + cálculo final --------------------------------------------
96 costo_peso = redondear(peso * tarifa_kg) # [N20]
97 subtotal = COSTO_BASE + costo_peso
98 descuento = redondear(subtotal * porcentaje_descuento)
99 recargo = redondear(subtotal * porcentaje_recargo)
100 total = subtotal - descuento + recargo
101
102 return {
103 "peso_kg": peso,
104 "tipo_cliente": str(tipo_cliente),
105 "tipo_envio": str(tipo_envio),
106 "costo_base": COSTO_BASE,
107 "tarifa_por_kg": tarifa_kg,
108 "costo_peso": costo_peso,

112 "porcentaje_recargo": porcentaje_recargo,
113 "recargo": recargo,
114 "total": total,
115 } # [N21] fin
